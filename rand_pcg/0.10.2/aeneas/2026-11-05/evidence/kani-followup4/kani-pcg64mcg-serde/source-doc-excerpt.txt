    /// - `state = 0xcafef00dd15ea5e5`
    pub fn new(state: u128) -> Self {
        // Force low bit to 1, as in C version (C++ uses `state | 3` instead).
        Mcg128Xsl64 { state: state | 1 }
    }

    /// Current state of the generator
    ///
    /// Will always be odd.
    pub fn state(&self) -> u128 {
        self.state
    }
}
