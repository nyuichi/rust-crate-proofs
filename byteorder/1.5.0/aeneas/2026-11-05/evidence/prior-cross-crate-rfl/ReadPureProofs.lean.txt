import ByteorderReadPureRoots.Funs

theorem be_read_u48_body (buf : Slice U8) : byteorder_read_pure_roots.byteorder.BigEndian.Insts.ByteorderByteOrder.read_u48 buf = byteorder_lite_u48_roots.byteorder_lite.BigEndian.Insts.Byteorder_liteByteOrder.read_u48 buf := by rfl
theorem be_read_u48_contract (buf : Slice U8) :
 ((6 ≤ buf.length) → byteorder_read_pure_roots.byteorder.BigEndian.Insts.ByteorderByteOrder.read_u48 buf ⦃ r => r.val = (BitVec.fromBEBytes ((buf.val.take 6).map U8.bv)).toNat ∧ r.val < 2^48 ⦄) ∧
 (buf.length < 6 → byteorder_read_pure_roots.byteorder.BigEndian.Insts.ByteorderByteOrder.read_u48 buf = .fail .assertionFailure) := by
 simpa only [be_read_u48_body] using U48Proofs.be_read_u48_final buf
theorem be_read_int_body (buf : Slice U8) (count : Usize) : byteorder_read_pure_roots.byteorder.BigEndian.Insts.ByteorderByteOrder.read_int buf count = signed_read_roots.byteorder_lite.BigEndian.Insts.Byteorder_liteByteOrder.read_int buf count := by rfl
theorem be_read_int_contract (buf : Slice U8) (count : Usize) :
    ((1 ≤ count.val ∧ count.val ≤ 8 ∧ count.val ≤ buf.length) →
      byteorder_read_pure_roots.byteorder.BigEndian.Insts.ByteorderByteOrder.read_int buf count
        ⦃ r => r.val = (BitVec.fromBEBytes ((buf.val.take count.val).map U8.bv)).toInt ⦄) ∧
    (¬ (1 ≤ count.val ∧ count.val ≤ 8 ∧ count.val ≤ buf.length) →
      byteorder_read_pure_roots.byteorder.BigEndian.Insts.ByteorderByteOrder.read_int buf count = .fail .assertionFailure) := by
 simpa only [be_read_int_body] using SignedReadProofs.be_read_int_contract buf count
