pub struct TryGetError {
    /// The number of bytes necessary to get the value
    pub requested: usize,

    /// The number of bytes available in the buffer
    pub available: usize,
}

fn try_get_u8(&mut self) -> Result<u8, TryGetError> {
        crate::slice_read_ops::read_u8(self).ok_or(TryGetError {
            requested: 1,
            available: self.len(),
        })
    }

fn get_u8(&mut self) -> u8 {
        match self.try_get_u8() {
            Ok(value) => value,
            Err(error) => panic_advance(&error),
        }
    }

fn panic_advance(error_info: &TryGetError) -> ! {
    panic!(
        "advance out of bounds: the len is {} but advancing by {}",
        error_info.available, error_info.requested
    );
}

fn panic_does_not_fit(size: usize, nbytes: usize) -> ! {
    panic!(
        "size too large: the integer type can fit {} bytes, but nbytes is {}",
        size, nbytes
    );
}

fn remaining(&self) -> usize {
        self.len()
    }

fn chunk(&self) -> &[u8] {
        self
    }

fn try_get_uint(&mut self, nbytes: usize) -> Result<u64, TryGetError> {
        let _slice_at = match 8usize.checked_sub(nbytes) {
            Some(slice_at) => slice_at,
            None => panic_does_not_fit(8, nbytes),
        };

        crate::variable_read_ops::read_be_u64(self, nbytes).ok_or(TryGetError {
            requested: nbytes,
            available: self.len(),
        })
    }

fn get_uint(&mut self, nbytes: usize) -> u64 {
        match self.try_get_uint(nbytes) {
            Ok(value) => value,
            Err(error) => panic_advance(&error),
        }
    }
